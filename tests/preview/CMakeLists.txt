set(BOREALIS_LIBRARY ${CMAKE_CURRENT_SOURCE_DIR}/dependencies/nx-player/library/borealis/library)
include(${BOREALIS_LIBRARY}/cmake/commonOption.cmake)
set(PLATFORM_DESKTOP ON)
set(USE_GL3 ON)
set(USE_SYSTEM_GLFW OFF)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
set(USE_SYSTEM_FMT OFF)
set(USE_SYSTEM_TINYXML2 OFF)
set(USE_SYSTEM_TWEENY OFF)
set(BRLS_RESOURCES_DIR ${BOREALIS_LIBRARY}/..)
include(${BOREALIS_LIBRARY}/cmake/toolchain.cmake)
project(hongguo-preview LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory(${BOREALIS_LIBRARY} borealis)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(POSTER_CODECS REQUIRED IMPORTED_TARGET libavcodec libswscale)
set(QJS ${CMAKE_CURRENT_SOURCE_DIR}/dependencies/quickjs)
add_library(qjs STATIC ${QJS}/quickjs.c ${QJS}/dtoa.c ${QJS}/libregexp.c ${QJS}/libunicode.c ${QJS}/cutils.c)
target_compile_definitions(qjs PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_include_directories(qjs PUBLIC ${QJS})
add_executable(hongguo-preview native/main.cpp native/backend.cpp native/poster.cpp native/poster_decode.cpp native/workers.cpp)
target_compile_definitions(hongguo-preview PRIVATE HONGGUO_PREVIEW)
target_include_directories(hongguo-preview PRIVATE tests/preview native)
target_link_libraries(hongguo-preview PRIVATE borealis qjs PkgConfig::MPV PkgConfig::CURL PkgConfig::POSTER_CODECS mbedtls mbedx509 mbedcrypto pthread dl)
